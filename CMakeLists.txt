string(JSON static_count LENGTH "${static_registry}" tables)
math(EXPR static_last "${static_count} - 1")
set(static_call_tables "")
foreach(index RANGE ${static_last})
    foreach(field IN ITEMS site proof_address prefix start end stride kind)
        string(JSON static_${field} GET "${static_registry}" tables ${index} ${field})
    endforeach()
    if(static_kind STREQUAL "jump")
        set(static_jump true)
        set(static_tail false)
    elseif(static_kind STREQUAL "tail")
        set(static_jump false)
        set(static_tail true)
    elseif(static_kind STREQUAL "call")
        set(static_jump false)
        set(static_tail false)
    else()
        message(FATAL_ERROR "Unknown static dispatch kind: ${static_kind}")
    endif()
    string(APPEND static_call_tables "    StaticCallTable{${static_site},${static_proof_address},\"${static_prefix}\",${static_start},${static_end},${static_stride},${static_jump},${static_tail},0},\n")
endforeach()

        string(JSON entry_${field} GET "${static_registry}" entry_tables ${index} ${field})
    endforeach()
    string(JSON entry_stride ERROR_VARIABLE entry_stride_error GET "${static_registry}" entry_tables ${index} stride)
    if(entry_stride_error)

    endif()
    string(APPEND entry_tables "    EntryTable{${entry_owner},\"${entry_prefix}\",${entry_start},${entry_end},${entry_stride}},\n")
endforeach()
# Share checked target-literal locations across sites using the same slot.
string(JSON group_count LENGTH "${static_registry}" groups)
if(group_count GREATER 0)
    math(EXPR group_last "${group_count} - 1")
    foreach(index RANGE ${group_last})
        foreach(field IN ITEMS prefix kind slot)
            string(JSON group_${field} ERROR_VARIABLE group_error GET "${static_registry}" groups ${index} ${field})
            if(group_error)
                message(FATAL_ERROR "Invalid grouped dispatch ${index}: ${group_error}")
            endif()
        endforeach()
        if(group_kind STREQUAL "call")
            set(group_tail false)
        elseif(group_kind STREQUAL "tail")
            set(group_tail true)
        else()
            message(FATAL_ERROR "Unknown grouped dispatch kind: ${group_kind}")
        endif()
        string(JSON site_count ERROR_VARIABLE group_error LENGTH "${static_registry}" groups ${index} sites)
        if(group_error)
            message(FATAL_ERROR "Invalid grouped dispatch ${index}: ${group_error}")
        endif()
        string(JSON target_count ERROR_VARIABLE group_error LENGTH "${static_registry}" groups ${index} target_literals)
        if(group_error)
            message(FATAL_ERROR "Invalid grouped dispatch ${index}: ${group_error}")
        endif()
        if(site_count EQUAL 0 OR target_count EQUAL 0)
            message(FATAL_ERROR "Grouped dispatch requires sites and target literals.")
        endif()
        math(EXPR site_last "${site_count} - 1")
        math(EXPR target_last "${target_count} - 1")
        foreach(site_index RANGE ${site_last})
            string(JSON group_site GET "${static_registry}" groups ${index} sites ${site_index})
            foreach(target_index RANGE ${target_last})
                string(JSON group_literal GET "${static_registry}" groups ${index} target_literals ${target_index})
                math(EXPR group_end "${group_literal} + 4" OUTPUT_FORMAT HEXADECIMAL)
                string(APPEND static_call_tables "    StaticCallTable{${group_site},${group_site},\"${group_prefix}\",${group_literal},${group_end},4,false,${group_tail},${group_slot}},\n")
            endforeach()
            string(JSON fault_count ERROR_VARIABLE fault_error LENGTH "${static_registry}" groups ${index} fault_literals)
            if(NOT fault_error AND fault_count GREATER 0)
                if(fault_count GREATER 4096)
                    message(FATAL_ERROR "Too many null callback literals")
                endif()
                math(EXPR fault_last "${fault_count} - 1")
                foreach(fault_index RANGE ${fault_last})
                    string(JSON group_literal GET "${static_registry}" groups ${index} fault_literals ${fault_index})
                    math(EXPR group_end "${group_literal} + 4" OUTPUT_FORMAT HEXADECIMAL)
                    string(APPEND static_call_tables "    StaticCallTable{${group_site},${group_site},\"${group_prefix}\",${group_literal},${group_end},4,false,${group_tail},${group_slot},true},\n")
                endforeach()
            endif()
        endforeach()
    endforeach()
endif()

string(JSON pointer_record_count LENGTH "${static_registry}" pointer_sources)
set(pointer_source_count 0)
set(table_pointer_sources "")
if(pointer_record_count GREATER 0)
    math(EXPR pointer_source_last "${pointer_record_count} - 1")
    foreach(index RANGE ${pointer_source_last})
        foreach(field IN ITEMS prefix kind)
            string(JSON pointer_${field} GET "${static_registry}" pointer_sources ${index} ${field})
        endforeach()
        foreach(field IN ITEMS source table)
            string(JSON pointer_${field} ERROR_VARIABLE pointer_error GET "${static_registry}" pointer_sources ${index} ${field})
            set(pointer_${field}_values "")
            if(pointer_error)
                string(JSON pointer_count LENGTH "${static_registry}" pointer_sources ${index} ${field}s)
                if(pointer_count LESS 1 OR pointer_count GREATER 4096)
                    message(FATAL_ERROR "Invalid pointer ${field} list")
                endif()
                math(EXPR pointer_last "${pointer_count} - 1")
                foreach(item RANGE ${pointer_last})
                    string(JSON pointer_value GET "${static_registry}" pointer_sources ${index} ${field}s ${item})
                    list(APPEND pointer_${field}_values "${pointer_value}")
                endforeach()
            else()
                list(APPEND pointer_${field}_values "${pointer_${field}}")
            endif()
        endforeach()
        string(JSON pointer_proof_address ERROR_VARIABLE pointer_proof_error GET "${static_registry}" pointer_sources ${index} proof_address)
        if(NOT pointer_kind MATCHES "^(return|global|load)-(table|object)$")
            message(FATAL_ERROR "Unknown table pointer source: ${pointer_kind}")
        endif()
        set(pointer_global false)
        set(pointer_object false)
        set(pointer_load false)
        if(pointer_kind MATCHES "^global-")
            set(pointer_global true)
        endif()
        if(pointer_kind MATCHES "-object$")
            set(pointer_object true)
        endif()
        if(pointer_kind MATCHES "^load-")
            set(pointer_load true)
        endif()
        list(LENGTH pointer_source_values pointer_sources_size)
        list(LENGTH pointer_table_values pointer_tables_size)
        math(EXPR pointer_expanded_count "${pointer_source_count} + ${pointer_sources_size} * ${pointer_tables_size}")
        if(pointer_expanded_count GREATER 4096)
            message(FATAL_ERROR "Too many pointer source bindings")
        endif()
        foreach(pointer_source IN LISTS pointer_source_values)
            if(pointer_proof_error)
                set(pointer_proof_address "${pointer_source}")
            endif()
            foreach(pointer_table IN LISTS pointer_table_values)
                string(APPEND table_pointer_sources "    TablePointerSource{${pointer_source},${pointer_proof_address},\"${pointer_prefix}\",${pointer_table},${pointer_global},${pointer_object},${pointer_load}},\n")
                math(EXPR pointer_source_count "${pointer_source_count} + 1")
            endforeach()
        endforeach()
    endforeach()
endif()
if(pointer_source_count GREATER 4096)
    message(FATAL_ERROR "Too many pointer source bindings")
endif()

if(TARGET smbd-cpu AND SMBD_CPU_PROFILE)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "CPU profiling requires Clang")
    endif()
    if(SMBD_CPU_PROFILE STREQUAL "GENERATE")
        execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-resource-dir
            OUTPUT_VARIABLE clang_resources OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        set(profile_runtime "${clang_resources}/lib/windows/clang_rt.profile-x86_64.lib")
        if(NOT EXISTS "${profile_runtime}")
            message(FATAL_ERROR "Clang's CPU profiling runtime is missing: ${profile_runtime}")
        endif()
        target_compile_options(smbd-cpu PRIVATE "$<$<CONFIG:Release>:-fprofile-generate=${CMAKE_CURRENT_BINARY_DIR}/cpu-profile>")
        target_link_libraries(smbd-cpu INTERFACE "${profile_runtime}")
        target_link_options(smbd-cpu INTERFACE "$<$<CONFIG:Release>:/include:__llvm_profile_runtime>")
    else()
        if(NOT EXISTS "${SMBD_CPU_PROFILE}" OR IS_DIRECTORY "${SMBD_CPU_PROFILE}")
            message(FATAL_ERROR "CPU profile file is missing: ${SMBD_CPU_PROFILE}")
        endif()
        file(REAL_PATH "${SMBD_CPU_PROFILE}" cpu_profile)
        target_compile_options(smbd-cpu PRIVATE "$<$<CONFIG:Release>:-fprofile-use=${cpu_profile};-Werror=profile-instr-out-of-date>")
        set_property(SOURCE "${SMBD_BATCH}" ${shards} src/cpu.cpp src/floating.cpp
            APPEND PROPERTY OBJECT_DEPENDS "${cpu_profile}")
    endif()
endif()
