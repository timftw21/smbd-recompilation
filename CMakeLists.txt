file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/kernel_imports.json" imports)
string(JSON import_count LENGTH "${imports}")
math(EXPR import_last "${import_count} - 1")
set(kernel_names "")
foreach(index RANGE ${import_last})
    string(JSON ordinal GET "${imports}" ${index} ordinal)
    string(JSON name GET "${imports}" ${index} name)
    string(APPEND kernel_names "    case ${ordinal}: return \"${name}\";\n")
endforeach()
configure_file(cmake/kernel_names.h.in generated/kernel_names.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/callback_abis.json" callback_registry)
string(JSON callback_count LENGTH "${callback_registry}" contracts)
math(EXPR callback_last "${callback_count} - 1")
set(callback_abis "")
foreach(index RANGE ${callback_last})
    foreach(field IN ITEMS site address ordinal argument_count prefix)
        string(JSON cb_${field} GET "${callback_registry}" contracts ${index} ${field})
    endforeach()
    string(JSON cb_count LENGTH "${callback_registry}" contracts ${index} callbacks)
    math(EXPR cb_last "${cb_count} - 1")
    set(cb_slots "")
    set(cb_offsets "")
    foreach(slot RANGE ${cb_last})
        string(JSON value GET "${callback_registry}" contracts ${index} callbacks ${slot})
        string(APPEND cb_slots "${value},")
        string(JSON offset GET "${callback_registry}" contracts ${index} offsets ${slot})
        string(APPEND cb_offsets "${offset},")
    endforeach()
    string(APPEND callback_abis "    CallbackAbi{${cb_site},${cb_address},${cb_ordinal},${cb_argument_count},\"${cb_prefix}\",{${cb_slots}},{${cb_offsets}},${cb_count}},\n")
endforeach()
string(JSON preserver_count LENGTH "${callback_registry}" preservers)
math(EXPR preserver_last "${preserver_count} - 1")
set(callback_preservers "")
foreach(index RANGE ${preserver_last})
    foreach(field IN ITEMS address mask prefix)
        string(JSON preserver_${field} GET "${callback_registry}" preservers ${index} ${field})
    endforeach()
    string(APPEND callback_preservers "    CallbackPreserver{${preserver_address},${preserver_mask},\"${preserver_prefix}\"},\n")
endforeach()
string(JSON factory_count LENGTH "${callback_registry}" factories)
math(EXPR factory_last "${factory_count} - 1")
set(callback_factories "")
foreach(index RANGE ${factory_last})
    string(JSON factory_address GET "${callback_registry}" factories ${index} address)
    string(JSON factory_prefix GET "${callback_registry}" factories ${index} prefix)
    string(JSON field_count LENGTH "${callback_registry}" factories ${index} fields)
    math(EXPR field_last "${field_count} - 1")
    set(factory_fields "")
    foreach(field RANGE ${field_last})
        string(JSON value GET "${callback_registry}" factories ${index} fields ${field})
        string(APPEND factory_fields "${value},")
    endforeach()
    string(APPEND callback_factories "    CallbackFactory{${factory_address},\"${factory_prefix}\",{${factory_fields}},${field_count}},\n")
endforeach()
configure_file(cmake/callback_abis.h.in generated/callback_abis.h @ONLY)

configure_file(cmake/native_apis.h.in generated/native_apis.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/static_calls.json" static_registry)
string(JSON static_sha GET "${static_registry}" executable_sha256)
if(NOT static_sha STREQUAL SMBD_file_sha256)
    message(FATAL_ERROR "Static call evidence belongs to another executable.")
endif()
string(JSON static_count LENGTH "${static_registry}" tables)
math(EXPR static_last "${static_count} - 1")
set(static_call_tables "")
foreach(index RANGE ${static_last})
    foreach(field IN ITEMS site proof_address prefix start end stride kind)
        string(JSON static_${field} GET "${static_registry}" tables ${index} ${field})
    endforeach()
    if(static_kind STREQUAL "jump")
        set(static_jump true)
        set(static_tail false)
    elseif(static_kind STREQUAL "tail")
        set(static_jump false)
        set(static_tail true)
    elseif(static_kind STREQUAL "call")
        set(static_jump false)
        set(static_tail false)
    else()
        message(FATAL_ERROR "Unknown static dispatch kind: ${static_kind}")
    endif()
    string(APPEND static_call_tables "    StaticCallTable{${static_site},${static_proof_address},\"${static_prefix}\",${static_start},${static_end},${static_stride},${static_jump},${static_tail},0},\n")
endforeach()
string(JSON entry_count LENGTH "${static_registry}" entry_tables)
math(EXPR entry_last "${entry_count} - 1")
set(entry_tables "")
foreach(index RANGE ${entry_last})
    foreach(field IN ITEMS owner prefix start end)
        string(JSON entry_${field} GET "${static_registry}" entry_tables ${index} ${field})
    endforeach()
    string(JSON entry_stride ERROR_VARIABLE entry_stride_error GET "${static_registry}" entry_tables ${index} stride)
    if(entry_stride_error)
        set(entry_stride 4)
    endif()
    string(APPEND entry_tables "    EntryTable{${entry_owner},\"${entry_prefix}\",${entry_start},${entry_end},${entry_stride}},\n")
endforeach()
# Share checked target-literal locations across sites using the same slot.
string(JSON group_count LENGTH "${static_registry}" groups)
if(group_count GREATER 0)
    math(EXPR group_last "${group_count} - 1")
    foreach(index RANGE ${group_last})
        foreach(field IN ITEMS prefix kind slot)
            string(JSON group_${field} GET "${static_registry}" groups ${index} ${field})
        endforeach()
        if(group_kind STREQUAL "call")
            set(group_tail false)
        elseif(group_kind STREQUAL "tail")
            set(group_tail true)
        else()
            message(FATAL_ERROR "Unknown grouped dispatch kind: ${group_kind}")
        endif()
        string(JSON site_count LENGTH "${static_registry}" groups ${index} sites)
        string(JSON target_count LENGTH "${static_registry}" groups ${index} target_literals)
        if(site_count EQUAL 0 OR target_count EQUAL 0)
            message(FATAL_ERROR "Grouped dispatch requires sites and target literals.")
        endif()
        math(EXPR site_last "${site_count} - 1")
        math(EXPR target_last "${target_count} - 1")
        foreach(site_index RANGE ${site_last})
            string(JSON group_site GET "${static_registry}" groups ${index} sites ${site_index})
            foreach(target_index RANGE ${target_last})
                string(JSON group_literal GET "${static_registry}" groups ${index} target_literals ${target_index})
                math(EXPR group_end "${group_literal} + 4" OUTPUT_FORMAT HEXADECIMAL)
                string(APPEND static_call_tables "    StaticCallTable{${group_site},${group_site},\"${group_prefix}\",${group_literal},${group_end},4,false,${group_tail},${group_slot}},\n")
            endforeach()
        endforeach()
    endforeach()
endif()

if(SMBD_BATCH)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Compiling native CPU batches currently requires Windows x64.")
    endif()
    enable_language(ASM_MASM)
    if(CMAKE_GENERATOR MATCHES "^Ninja")
        set(CMAKE_ASM_MASM_COMPILE_OBJECT "<CMAKE_ASM_MASM_COMPILER> <DEFINES> <INCLUDES> /c /Fo<OBJECT> <FLAGS>")
        foreach(assembly IN ITEMS "${SMBD_BATCH}.asm" "${CMAKE_CURRENT_SOURCE_DIR}/src/cpu_floating.asm")
            file(RELATIVE_PATH assembly_input "${CMAKE_CURRENT_BINARY_DIR}" "${assembly}")
            set_source_files_properties("${assembly}" PROPERTIES COMPILE_OPTIONS "/Ta${assembly_input}")
        endforeach()
    endif()
    file(GLOB shards CONFIGURE_DEPENDS
        "${SMBD_BATCH}.[0-9][0-9].cpp" "${SMBD_BATCH}.[0-9][0-9][0-9].cpp")
    if(MSVC)
        # Retain instruction-address labels for debugging generated code.
        set_source_files_properties(${shards} PROPERTIES COMPILE_OPTIONS /wd4102)
    endif()
    # Keep C++ dependency flags out of MASM's command line.
    add_library(smbd-helpers OBJECT "${SMBD_BATCH}.asm" src/cpu_floating.asm)
    add_library(smbd-cpu STATIC "${SMBD_BATCH}" ${shards} src/cpu.cpp src/floating.cpp)
    # Isolate changing registry data from ThinLTO imports in guest code.
    if(shards AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        set_property(SOURCE "${SMBD_BATCH}" APPEND PROPERTY COMPILE_OPTIONS "$<$<CONFIG:Release>:-fno-lto>")
    endif()
    target_precompile_headers(smbd-cpu PRIVATE <cpu.h>)
    if(CMAKE_GENERATOR MATCHES "^Ninja")
        set_property(TARGET smbd-cpu PROPERTY JOB_POOL_COMPILE cpu_compile)
    endif()
    target_link_libraries(smbd-cpu PUBLIC smbd-core PRIVATE smbd-helpers)
    target_include_directories(smbd-cpu PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(MSVC)
        target_compile_options(smbd-cpu PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
    endif()
    target_sources(smbd-tool PRIVATE src/cpu_diagnostics.cpp)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    set(SDL_GPU OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER OFF CACHE BOOL "" FORCE)
    set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
    set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
    set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(sdl3
        URL https://github.com/libsdl-org/SDL/archive/829a65d769d935c4852f8159e964312c0957260a.tar.gz
        URL_HASH SHA256=48E15CF1A6928453D77AFA955D1FDFBBCA9169E2BC3D76D111ACB9B8BB076D66)
    FetchContent_MakeAvailable(sdl3)
    add_library(smbd-video STATIC src/render.cpp src/gpu.cpp)
    target_link_libraries(smbd-video PUBLIC smbd-shaders PRIVATE d3d11 dxgi windowscodecs ole32)
    add_library(smbd-input STATIC src/input.cpp)
    target_link_libraries(smbd-input PUBLIC smbd-core SDL3::SDL3-static)
    add_library(smbd-audio STATIC src/audio.cpp)
    target_link_libraries(smbd-audio PUBLIC smbd-core SDL3::SDL3-static)
    add_library(smbd-kernel STATIC src/xbox.cpp src/boot.cpp)
    target_include_directories(smbd-kernel PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(smbd-kernel PUBLIC smbd-core smbd-video smbd-input smbd-audio PRIVATE iphlpapi)
    if(MSVC)
        target_compile_options(smbd-kernel PRIVATE /W4 /utf-8)
    endif()
    target_link_libraries(smbd-tool PRIVATE smbd-cpu smbd-kernel)
    target_compile_definitions(smbd-tool PRIVATE SMBD_HAS_BATCH=1)
    if(SMBD_SHADER_BANK)
