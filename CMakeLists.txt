cmake_minimum_required(VERSION 4.2)
project(smbd VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/supported_target.json" registry)
foreach(field IN ITEMS target_id file_size file_sha256)
    string(JSON SMBD_${field} GET "${registry}" targets 0 ${field})
endforeach()
configure_file(cmake/target.h.in generated/target.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/kernel_imports.json" imports)
string(JSON import_count LENGTH "${imports}")
math(EXPR import_last "${import_count} - 1")
set(kernel_names "")
foreach(index RANGE ${import_last})
    string(JSON ordinal GET "${imports}" ${index} ordinal)
    string(JSON name GET "${imports}" ${index} name)
    string(APPEND kernel_names "    case ${ordinal}: return \"${name}\";\n")
endforeach()
configure_file(cmake/kernel_names.h.in generated/kernel_names.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/callback_abis.json" callback_registry)
string(JSON callback_count LENGTH "${callback_registry}" contracts)
math(EXPR callback_last "${callback_count} - 1")
set(callback_abis "")
foreach(index RANGE ${callback_last})
    foreach(field IN ITEMS site address ordinal argument_count prefix)
        string(JSON cb_${field} GET "${callback_registry}" contracts ${index} ${field})
    endforeach()
    string(JSON cb_count LENGTH "${callback_registry}" contracts ${index} callbacks)
    math(EXPR cb_last "${cb_count} - 1")
    set(cb_slots "")
    set(cb_offsets "")
    foreach(slot RANGE ${cb_last})
        string(JSON value GET "${callback_registry}" contracts ${index} callbacks ${slot})
        string(APPEND cb_slots "${value},")
        string(JSON offset GET "${callback_registry}" contracts ${index} offsets ${slot})
        string(APPEND cb_offsets "${offset},")
    endforeach()
    string(APPEND callback_abis "    CallbackAbi{${cb_site},${cb_address},${cb_ordinal},${cb_argument_count},\"${cb_prefix}\",{${cb_slots}},{${cb_offsets}},${cb_count}},\n")
endforeach()
configure_file(cmake/callback_abis.h.in generated/callback_abis.h @ONLY)
string(JSON cb_sha GET "${callback_registry}" executable_sha256)
if(NOT cb_sha STREQUAL SMBD_file_sha256)
    message(FATAL_ERROR "Callback ABI registry belongs to another executable.")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/native_apis.json" api_registry)
string(JSON api_sha GET "${api_registry}" executable_sha256)
if(NOT api_sha STREQUAL SMBD_file_sha256)
    message(FATAL_ERROR "Native API registry belongs to another executable.")
endif()
string(JSON api_count LENGTH "${api_registry}" apis)
math(EXPR api_last "${api_count} - 1")
set(native_apis "")
foreach(index RANGE ${api_last})
    foreach(field IN ITEMS address name prefix stack_cleanup)
        string(JSON api_${field} GET "${api_registry}" apis ${index} ${field})
    endforeach()
    string(APPEND native_apis "    NativeApi{${api_address},\"${api_name}\",\"${api_prefix}\",${api_stack_cleanup}},\n")
endforeach()
configure_file(cmake/native_apis.h.in generated/native_apis.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/knowledge/static_calls.json" static_registry)
string(JSON static_sha GET "${static_registry}" executable_sha256)
if(NOT static_sha STREQUAL SMBD_file_sha256)
    message(FATAL_ERROR "Static call evidence belongs to another executable.")
endif()
string(JSON static_count LENGTH "${static_registry}" tables)
math(EXPR static_last "${static_count} - 1")
set(static_call_tables "")
foreach(index RANGE ${static_last})
    foreach(field IN ITEMS site proof_address prefix start end stride kind)
        string(JSON static_${field} GET "${static_registry}" tables ${index} ${field})
    endforeach()
    if(static_kind STREQUAL "jump")
        set(static_jump true)
    else()
        set(static_jump false)
    endif()
    string(APPEND static_call_tables "    StaticCallTable{${static_site},${static_proof_address},\"${static_prefix}\",${static_start},${static_end},${static_stride},${static_jump}},\n")
endforeach()
string(JSON entry_count LENGTH "${static_registry}" entry_tables)
math(EXPR entry_last "${entry_count} - 1")
set(entry_tables "")
foreach(index RANGE ${entry_last})
    foreach(field IN ITEMS owner prefix start end)
        string(JSON entry_${field} GET "${static_registry}" entry_tables ${index} ${field})
    endforeach()
    string(APPEND entry_tables "    EntryTable{${entry_owner},\"${entry_prefix}\",${entry_start},${entry_end}},\n")
endforeach()
configure_file(cmake/static_calls.h.in generated/static_calls.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    knowledge/supported_target.json knowledge/kernel_imports.json knowledge/callback_abis.json knowledge/native_apis.json knowledge/static_calls.json)

include(FetchContent)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
set(ZYDIS_FEATURE_ENCODER OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zydis
    GIT_REPOSITORY https://github.com/zyantific/zydis.git
    GIT_TAG a2278f1d254e492f6a6b39f6cb5d1f5d515659dc
    GIT_SUBMODULES dependencies/zycore)
FetchContent_MakeAvailable(zydis)

add_library(smbd-core STATIC src/file.cpp src/disc.cpp src/xbe.cpp src/crypto.cpp)
target_include_directories(smbd-core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(WIN32)
    target_compile_definitions(smbd-core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_link_libraries(smbd-core PRIVATE bcrypt)
else()
    find_package(OpenSSL REQUIRED)
    target_link_libraries(smbd-core PRIVATE OpenSSL::Crypto)
endif()

add_executable(smbd-tool src/tool.cpp src/pipeline.cpp src/decoder.cpp src/recompiler.cpp src/batch.cpp)
target_include_directories(smbd-tool PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(smbd-tool PRIVATE smbd-core Zydis)
if(MSVC)
    target_compile_options(smbd-core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
    target_compile_options(smbd-tool PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
endif()

if(WIN32)
    add_library(smbd-shaders STATIC src/shaders.cpp src/vertex_program.cpp)
    target_link_libraries(smbd-shaders PUBLIC smbd-core)
    target_sources(smbd-tool PRIVATE src/shader_compiler.cpp)
    target_link_libraries(smbd-tool PRIVATE smbd-shaders d3dcompiler)
    target_compile_definitions(smbd-tool PRIVATE SMBD_HAS_SHADERS=1)
endif()

set(SMBD_BATCH "" CACHE FILEPATH "Local generated batch .cpp to compile into offline diagnostics")
if(SMBD_BATCH)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Compiling native CPU batches currently requires Windows x64.")
    endif()
    enable_language(ASM_MASM)
    file(GLOB shards CONFIGURE_DEPENDS "${SMBD_BATCH}.[0-9][0-9].cpp")
    if(MSVC)
        # Retain instruction-address labels for debugging generated code.
        set_source_files_properties(${shards} PROPERTIES COMPILE_OPTIONS /wd4102)
    endif()
    # Keep C++ dependency flags out of MASM's command line.
    add_library(smbd-helpers OBJECT "${SMBD_BATCH}.asm" src/cpu_floating.asm)
    add_library(smbd-cpu STATIC "${SMBD_BATCH}" ${shards} src/cpu.cpp src/floating.cpp)
    target_link_libraries(smbd-cpu PUBLIC smbd-core PRIVATE smbd-helpers)
    target_include_directories(smbd-cpu PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(MSVC)
        target_compile_options(smbd-cpu PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
    endif()
    target_sources(smbd-tool PRIVATE src/cpu_diagnostics.cpp)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    set(SDL_GPU OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER OFF CACHE BOOL "" FORCE)
    set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
    set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
    set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(sdl3
        URL https://github.com/libsdl-org/SDL/archive/829a65d769d935c4852f8159e964312c0957260a.tar.gz
        URL_HASH SHA256=48E15CF1A6928453D77AFA955D1FDFBBCA9169E2BC3D76D111ACB9B8BB076D66)
    FetchContent_MakeAvailable(sdl3)
    add_library(smbd-video STATIC src/render.cpp src/gpu.cpp)
    target_link_libraries(smbd-video PUBLIC smbd-shaders PRIVATE d3d11 dxgi windowscodecs ole32)
    add_library(smbd-input STATIC src/input.cpp)
    target_link_libraries(smbd-input PUBLIC smbd-core SDL3::SDL3-static)
    add_library(smbd-audio STATIC src/audio.cpp)
    target_link_libraries(smbd-audio PUBLIC smbd-core SDL3::SDL3-static)
    add_library(smbd-kernel STATIC src/xbox.cpp src/boot.cpp)
    target_include_directories(smbd-kernel PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(smbd-kernel PUBLIC smbd-core smbd-video smbd-input smbd-audio PRIVATE iphlpapi)
    if(MSVC)
        target_compile_options(smbd-kernel PRIVATE /W4 /utf-8)
    endif()
    target_link_libraries(smbd-tool PRIVATE smbd-cpu smbd-kernel)
    target_compile_definitions(smbd-tool PRIVATE SMBD_HAS_BATCH=1)
    set(SMBD_SHADER_BANK "" CACHE FILEPATH "Ahead-of-time generated shader bank for the game")
    if(SMBD_SHADER_BANK)
        add_library(smbd-shader-bank STATIC "${SMBD_SHADER_BANK}")
        target_link_libraries(smbd-shader-bank PUBLIC smbd-core)
        target_compile_definitions(smbd-tool PRIVATE SMBD_COMPILED_SHADERS=1)
        target_link_libraries(smbd-tool PRIVATE smbd-shader-bank)
        add_executable(smbd src/main.cpp)
        target_link_libraries(smbd PRIVATE smbd-cpu smbd-kernel smbd-shader-bank)
    endif()
endif()
